# Formato de los archivos de Dom32Touch
# =====================================
#
# Al inicio y al presionar un botón HOME se carga el archivo home.csv
# NEXT y PREV navegan por screen0.csv a screen9.csv o menu0.csv a menu9.csv
# 
# <-- Línea en blanco y comenzada con # se ignora
#
# FONDO,[color]
# BOTON_CUADRADO,[etiqueta],[comando:objeto],[x],[y],[w],[h],[color fondo],[color borde],[color etiqueta],[icono],[orientacion]
# BOTON_REDONDO,[etiqueta],[comando:objeto],[x],[y],[w],[h],[color fondo],[color borde],[color etiqueta],[icono],[orientacion]
#
# etiqueta: Texto que aparece en el botón (opcopnal)
# comando: SWITCH, PULSE, CONFIG, HOME, NEXT, PREV
# objeto:
# x: posición X de la esquina superior derecha del botón
# y: posición Y de la esquina superior derecha del botón
# w: Ancho del botón
# h: altura del botón
# color fondo: 0 a 65535
# color borde: 0 a 65535
# color etiqueta: 0 a 65535
# icono: archivo BMP de 48x48 pixel (opcopnal)
# orientacion: 0 = Horizontal, 1 = Vertical
#
# Colores
# BLACK        0
# BLUE         31
# RED          63488
# GREEN        2016
# CYAN         2047
# YELLOW       65504
# WHITE        65535
# NAVY         15
# DARKGREEN    992
# DARKCYAN     1007
# MAROON       3020
# PURPLE       30735
# OLIVE        31712
# LIGHTGREY    50712
# DARKGREY     31727
# ORANGE       64800
# GREENYELLOW  45029
# PINK         65519
#

Colores 16bit
https://github.com/newdigate/rgb565_colors


Comandos:

download Touch-POC, /download/touch/config.csv


configurar <dispositivo>, <parametro=valor>

Parametros generales:
ap1
ap1p
ap2
ap2p
ce1
ce2
path
ce1p
ce2p
report

Parametros touch:
brillo-max
brillo-med
brillo-min
screen-saver

configurar Touch-POC, brillo-min=10
configurar Touch-POC, screen-saver=5
